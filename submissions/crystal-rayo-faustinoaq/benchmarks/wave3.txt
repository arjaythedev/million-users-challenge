#!/usr/bin/env bash
set -euo pipefail
DIR=/home/user/experiments/crystal-challenges/rw-million/submissions/crystal-rayo-faustinoaq
REF=/home/user/experiments/crystal-challenges/million-users-challenge
prefix=rayo_million_w3_1028
pid=
db=
q() { docker exec mc1 su postgres -c "psql -p 5501 -v ON_ERROR_STOP=1 -qX -d postgres $*"; }
stop() {
  if [ -n "$pid" ]; then
    docker exec mc1 su ubuntu -c "kill $pid" || true
    pid=
    sleep 1
  fi
}
cleanup() {
  stop
  if [ -n "$db" ]; then q "-c 'DROP DATABASE IF EXISTS $db'"; fi
}
trap cleanup EXIT
for entry in epoll rayo; do
  for workload in feed users; do
    db=${prefix}_${entry}_${workload}
    q "-c 'CREATE DATABASE $db TEMPLATE rayo_million_w3_seed_1028'"
    binary=bin/server
    [ "$entry" != epoll ] || binary=local/server-epoll
    rm -f "$DIR/local/wave3-server.pid"
    docker exec -d mc1 su ubuntu -c "cd $DIR; echo \$\$ > local/wave3-server.pid; export DATABASE_URL=postgres://postgres:postgres@127.0.0.1:5501/$db JWT_SECRET=million-users-challenge HOST=127.0.0.1 PORT=18470 EXTRA_PORT=18471 WORKERS=3; exec taskset -c 0-2 $binary > local/wave3-$entry-$workload-server.log 2>&1"
    for i in $(seq 60); do
      if [ -s "$DIR/local/wave3-server.pid" ]; then pid=$(cat "$DIR/local/wave3-server.pid"); fi
      if curl -sf --max-time 1 http://127.0.0.1:18470/health >/dev/null && curl -sf --max-time 1 http://127.0.0.1:18471/health >/dev/null; then break; fi
      sleep 0.5
    done
    curl -sf http://127.0.0.1:18470/health
    echo
    echo "=== $entry $workload ==="
    if [ "$workload" = feed ]; then
      docker exec crdev su ubuntu -c "taskset -c 9-11 wrk -t3 -c256 -d10s --latency -H 'Accept-Encoding: gzip' http://127.0.0.1:18470/feed"
    else
      docker exec crdev su ubuntu -c "taskset -c 9-11 $DIR/local/mload -target 127.0.0.1 -ports 18470,18471 -users 3000 -ramp 5s -hold 15s -gzip -think-scale 0.02 -tokens $REF/seed/tokens.json -out $DIR/local/wave3-$entry-users.json"
    fi
    stop
    docker exec mc1 su postgres -c "psql -p 5501 -d conduit_seed -X -P pager=off -c \"SELECT calls, round(total_exec_time::numeric,1) AS exec_ms, round(mean_exec_time::numeric,3) AS mean_ms, (shared_blks_hit+shared_blks_read)/greatest(calls,1) AS blocks_per_call, left(regexp_replace(query,'\\s+',' ','g'),130) AS query FROM pg_stat_statements s JOIN pg_database d ON d.oid=s.dbid WHERE d.datname='$db' ORDER BY total_exec_time DESC\""
    q "-c 'DROP DATABASE $db'"
    db=
  done
done
