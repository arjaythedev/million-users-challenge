% feed.tex -- the whole application of the feed API, in plain e-TeX (pdfTeX -ini -etex -8bit).
%
% A long-lived worker: pdftex is started once from a precompiled format (see build-fmt.tex) with
% '\def\feedctl{<control dir>/}\feedloop'.  \feedloop reads one line at a time from the FIFO <control dir>/job.fifo;
% the line is a path prefix P ("" when the host's working directory for this worker is the request directory).
% The host has put P meta.txt / P req.txt / P body.txt in place.  \feed@serve parses the request, verifies the JWT
% (crypto: sha256.tex), parses the JSON body, sends one database command through P db.req / P db.resp and writes
% P resp.txt (status line, JSON line); \feed@respond then writes "ok" to <control dir>/done.fifo and unwinds to
% the loop.  Nothing is kept between requests: see "Isolation" below.
%
% Isolation: a request runs inside \begingroup...\endgroup, so every local assignment (registers, macros) is undone
% when it ends.  The few global assignments (\hmacresult, \bdecoded, \sharesult, \ifbvalid, \feed@jerr,
% \feed@bcnt, \feed@auth, \feed@T:*, \feed@V:*) are reset by \feed@scrub before and after every request.  No
% control sequence name is created from request data (\csname is only used with names that exist or from a
% fixed set), so the hash table does not grow.
%
% Conventions:
%  * Request data are strings of catcode-12 characters (spaces: catcode 10).
%  * A decoded JSON string is a token list of catcode-12 character tokens, in which a space is the
%    unexpandable token \feed@S (\chardef = 32), so that it survives being passed as a macro argument.
%  * Output is built with \edef from character tokens and the lookup tables \feed@B<n> (raw byte),
%    \feed@E<n> (byte as it must appear inside a JSON string) and \feed@h<xy> (hex pair -> \feed@E).
%  * Source lines are glued (\endlinechar=-1): there are no accidental spaces.

\catcode`\{=1 \catcode`\}=2 \catcode`\#=6 \catcode`\@=11
\ifx\feeddir\undefined\def\feeddir{}\fi
\input \feeddir sha256.tex
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6 \catcode`\@=11
\endlinechar=-1
\newlinechar=-1

  \else\ifnum\feed@i=9 \expandafter\xdef\csname feed@E9\endcsname{\feed@b{92}\feed@b{116}}%
  \else\ifnum\feed@i=10 \expandafter\xdef\csname feed@E10\endcsname{\feed@b{92}\feed@b{110}}%
  \else\ifnum\feed@i=12 \expandafter\xdef\csname feed@E12\endcsname{\feed@b{92}\feed@b{102}}%
  \else\ifnum\feed@i=13 \expandafter\xdef\csname feed@E13\endcsname{\feed@b{92}\feed@b{114}}%

% division helpers for UTF-8: \feed@x -> \feed@q = x div 64, \feed@r = x mod 64
\def\feed@dm{\feed@q=\feed@x \divide\feed@q 64 \relax
  \feed@r=\feed@q \multiply\feed@r 64 \feed@r=-\feed@r \advance\feed@r\feed@x }
\def\feed@utf{% emit code point \feed@cp as UTF-8 bytes
  \ifnum\feed@cp<1 \else
  \ifnum\feed@cp<128 \feed@appb\feed@cp \relax
  \else\ifnum\feed@cp<2048 \relax
    \feed@x=\feed@cp \feed@dm \feed@ta=\feed@q \feed@tb=\feed@r \relax
    \advance\feed@ta 192 \feed@appb\feed@ta \advance\feed@tb 128 \feed@appb\feed@tb \relax
  \else\ifnum\feed@cp<65536 \relax
    \feed@x=\feed@cp \feed@dm \feed@tc=\feed@r \feed@x=\feed@q \feed@dm
    \feed@ta=\feed@q \feed@tb=\feed@r \relax
    \advance\feed@ta 224 \feed@appb\feed@ta \advance\feed@tb 128 \feed@appb\feed@tb \relax
    \advance\feed@tc 128 \feed@appb\feed@tc \relax
  \else
    \feed@x=\feed@cp \feed@dm \feed@hi=\feed@r \feed@x=\feed@q \feed@dm \feed@tc=\feed@r \relax
    \feed@x=\feed@q \feed@dm \feed@ta=\feed@q \feed@tb=\feed@r \relax
    \advance\feed@ta 240 \feed@appb\feed@ta \advance\feed@tb 128 \feed@appb\feed@tb \relax
    \advance\feed@tc 128 \feed@appb\feed@tc \advance\feed@hi 128 \feed@appb\feed@hi \relax
  \fi\fi\fi\fi}

\def\feed@badtoken{\feed@err{401}{invalid or expired token}}

\def\feed@auth@need{%
  \ifx\feed@auth\feed@unset \feed@missing\fi
  \edef\feed@tmp{\feed@auth\space\space}%
  \expandafter\feed@brA\feed@tmp\feed@end
  \ifnum\pdfstrcmp{\feed@w}{Bearer}=0 \else\feed@missing\fi
  \ifx\feed@jtoken\feed@empty \feed@missing\fi
  % split into three dot separated parts
  \expandafter\feed@jwt\feed@jtoken.\feed@X.\feed@X.\feed@X\feed@end
  % signature
  \edef\feed@msg{\feed@jh.\feed@jp}%
  \edef\feed@tmp{\noexpand\hmacsha256base64url{\feed@secret}{\feed@msg}}\feed@tmp
  \ifnum\pdfstrcmp{\hmacresult}{\feed@js}=0 \else\feed@badtoken\fi
  % header
  \edef\feed@tmp{\noexpand\base64urldecode{\feed@jh}}\feed@tmp
  \ifbvalid\else\feed@badtoken\fi
  \let\feed@pm\bdecoded \feed@parsemacro\feed@pm
  \if1\feed@jerr\feed@badtoken\fi
  \feed@typeis{alg}\feed@tS \else\feed@badtoken\fi
  \ifnum\pdfstrcmp{\csname feed@V:alg\endcsname}{HS256}=0 \else\feed@badtoken\fi
  % payload
  \edef\feed@tmp{\noexpand\base64urldecode{\feed@jp}}\feed@tmp
  \ifbvalid\else\feed@badtoken\fi
  \let\feed@pm\bdecoded \feed@parsemacro\feed@pm
  \if1\feed@jerr\feed@badtoken\fi
  \feed@checkexp
  \feed@checksub
  \feed@typeis{username}\feed@tS \else\feed@badpayload\fi

  \edef\feed@e{\csname feed@V:exp\endcsname}%

  \feed@len=0 \expandafter\feed@lenl\feed@now\feed@end\feed@tb=\feed@len \relax
  \feed@d=0 \relax
  \ifnum\feed@ta>\feed@tb \feed@d=1 \else
    \ifnum\feed@ta=\feed@tb \ifnum\pdfstrcmp{\feed@e}{\feed@now}>0 \feed@d=1 \fi\fi
  \fi
  \ifnum\feed@d=1 \else\feed@badtoken\fi}

\def\feed@checksub{%

  \expandafter\feed@digits\expandafter{\feed@e}%

  \ifnum\feed@nz=1 \else\feed@badpayload\fi
  \ifnum\feed@len>18 \feed@badpayload\fi
  \let\feed@uid\feed@e}

% \feed@respond ends the request by skipping everything up to \feed@reqend (the marker \feed@serve put after the
